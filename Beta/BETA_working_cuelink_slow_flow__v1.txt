// =====================================================================
// Helpers for execution-time monitoring
// =====================================================================
function getTodayKey_() {
  return Utilities.formatDate(new Date(), Session.getScriptTimeZone(), 'yyyy-MM-dd');
}

function msToHuman_(ms) {
  ms = Number(ms) || 0;
  const totalSeconds = Math.floor(ms / 1000);
  const minutes = Math.floor(totalSeconds / 60);
  const seconds = totalSeconds % 60;
  if (minutes > 0) return `${minutes}m ${seconds}s`;
  return `${seconds}s`;
}

function epochMsToHuman_(epochMs) {
  const n = Number(epochMs);
  if (!isFinite(n) || n <= 0) return '';
  return Utilities.formatDate(new Date(n), Session.getScriptTimeZone(), 'yyyy-MM-dd HH:mm:ss');
}

function recordExecutionRuntime_(handlerName, startEpochMs_) {
  // Never allow monitoring to break the main deal pipeline.
  try {
    const elapsedMs = Date.now() - Number(startEpochMs_);
    if (!isFinite(elapsedMs) || elapsedMs < 0) return;

    const props = PropertiesService.getScriptProperties();
    const lock = LockService.getScriptLock();

    const lockAcquired = lock.tryLock(10000);
    if (!lockAcquired) return;

    try {
      const todayKey = getTodayKey_();
      const lastResetDate = props.getProperty(PROP_EXEC_DAILY_DATE);

      // Reset daily counters on first run after date change
      if (lastResetDate !== todayKey) {
        props.setProperty(PROP_EXEC_DAILY_DATE, todayKey);
        props.setProperty(PROP_EXEC_DAILY_TOTAL_MS, '0');
      }

      const prevTotalMs = Number(props.getProperty(PROP_EXEC_DAILY_TOTAL_MS) || '0');
      const newTotalMs = prevTotalMs + elapsedMs;
      props.setProperty(PROP_EXEC_DAILY_TOTAL_MS, String(newTotalMs));

      // Store per-handler last runtime snapshots (useful for later expansion)
      props.setProperty(getExecPropKey_(`LAST_RUNTIME_AT_${handlerName}`), String(Date.now()));
      props.setProperty(getExecPropKey_(`LAST_RUNTIME_MS_${handlerName}`), String(elapsedMs));

      // Throttle writes to the sheet
      const now = Date.now();
      const lastWrite = Number(props.getProperty(PROP_EXEC_LAST_WRITE_AT) || '0');
      if (now - lastWrite < CONFIG_WRITE_MIN_INTERVAL_MS) return;

      props.setProperty(PROP_EXEC_LAST_WRITE_AT, String(now));

      // Best-effort: update the Configurations tab cells (if present)
      const ss = SpreadsheetApp.getActive();
      const sheet = ss.getSheetByName(CONFIG_SHEET_NAME);
      if (!sheet) return;

      // Daily total
      sheet.getRange(CONFIG_DAILY_RUNTIME_CELL_A1).setValue(msToHuman_(newTotalMs));

      // Per-trigger last runtime (elapsed + last-run timestamp)
      const cellA1 = (CONFIG_LAST_RUNTIME_CELL_BY_HANDLER && CONFIG_LAST_RUNTIME_CELL_BY_HANDLER[handlerName])
        ? CONFIG_LAST_RUNTIME_CELL_BY_HANDLER[handlerName]
        : null;
      if (cellA1) {
        const lastRunAtEpochMs = Date.now();
        const prettyTime = epochMsToHuman_(lastRunAtEpochMs);
        sheet.getRange(cellA1).setValue(`${msToHuman_(elapsedMs)} | ${prettyTime}`);
      }
    } finally {
      lock.releaseLock();
    }
  } catch (e) {
    // swallow
  }
}

const CUELINKS_API_BASE = 'https://developers.cuelinks.com/pub_api/v3';
const CUELINKS_PAGE_SIZE = 100;   // offers per page (API max: 500)
const CUELINKS_MAX_PAGES = 10;    // safety cap: 1,000 offers/run max, keeps well under Apps Script's 6-min execution limit

/**
 * Runs the intake pipeline for Cuelinks (real JSON API) and every enabled
 * XML source in DEAL_SOURCES. Wire THIS function to your time-driven
 * trigger — this is the entry point.
 *
 * Quota is enforced PER SOURCE PER CATEGORY (Cuelinks can add up to N Gym
 * deals AND EarnKaro can separately add up to N Gym deals in the same
 * run) — dedup against Raw_URL history still prevents any actual
 * duplicate product from being posted twice, even across sources.
 */
function runAllIntakes() {
  const __trackStart = Date.now();
  try {
    try {
      fetchCuelinksOffers();
    } catch (err) {
      logError_('runAllIntakes (Cuelinks)', err);
    }

    DEAL_SOURCES.forEach(source => {
      if (!source.enabled) return;
      try {
        fetchDealsFromSource_(source);
      } catch (err) {
        logError_(`runAllIntakes (${source.name})`, err);
        // one source failing never blocks the others
      }
    });
  } finally {
    recordExecutionRuntime_('runAllIntakes', __trackStart);
  }
}

/**
 * Pulls live offers from the real Cuelinks Offers API, de-dupes against
 * existing Raw_URL history, keyword-categorizes each item, and appends
 * fresh "Pending" rows (WITH real Original_Price/Deal_Price this time —
 * the API provides them directly) up to a per-category quota per run.
 *
 * Only accepts offers where both original_price and discount_price are
 * present and discount_price < original_price — an offer_type: "coupon"
 * with no numeric prices can't be posted by processRow_ anyway (it
 * requires both price columns), so filtering here avoids burning quota
 * slots on rows that would just end up as "Error: missing field".
 *
 * Does NOT touch processRow_ or the Telegram posting logic — this only
 * writes rows; the existing onEdit/sweep triggers pick them up normally.
 */
function fetchCuelinksOffers() {
  const sheet = SpreadsheetApp.getActive().getSheetByName(SHEET_NAME);

  const apiKey = PropertiesService.getScriptProperties().getProperty('CUELINKS_API_KEY');
  if (!apiKey) {
    logError_('fetchCuelinksOffers', new Error('CUELINKS_API_KEY not set in Script Properties'));
    return;
  }

  const existingLinks = getExistingRawUrls_(sheet);

  const quotaLimit = SYSTEM_SETTINGS.IS_SALE_MODE_ACTIVE
    ? INTAKE_QUOTA.saleMode
    : INTAKE_QUOTA.normal;
  const quotaUsed = { 'Gym': 0, 'Pets': 0, 'Vehicle': 0 };

  const rowsToAppend = [];
  let page = 1;
  let totalPages = 1;

  pageLoop:
  while (page <= totalPages && page <= CUELINKS_MAX_PAGES) {
    const allFull = Object.keys(quotaUsed).every(cat => quotaUsed[cat] >= quotaLimit);
    if (allFull) break;

    const url = `${CUELINKS_API_BASE}/offers?page=${page}&per_page=${CUELINKS_PAGE_SIZE}&sort=created_at&order=desc`;

    let json;
    try {
      const response = UrlFetchApp.fetch(url, {
        method: 'get',
        headers: { 'Authorization': 'Token ' + apiKey },
        muteHttpExceptions: true
      });
      const code = response.getResponseCode();
      if (code !== 200) {
        throw new Error(`Cuelinks API returned HTTP ${code}: ${response.getContentText().slice(0, 200)}`);
      }
      json = JSON.parse(response.getContentText());
    } catch (err) {
      logError_(`fetchCuelinksOffers page ${page}`, err);
      break; // stop paging on any fetch/parse failure, keep whatever we already collected
    }

    totalPages = (json.meta && json.meta.total_pages) || 1;
    const offers = json.data || [];

    for (const offer of offers) {
      const allFullInner = Object.keys(quotaUsed).every(cat => quotaUsed[cat] >= quotaLimit);
      if (allFullInner) break pageLoop;

      try {
        const title = offer.title;
        const link = offer.tracking_url;
        const description = offer.description || '';

        if (!title || !link) continue; // malformed offer, skip silently

        // Tier 1: real structured prices from the API (deal-type offers).
        // Tier 2: fall back to regex extraction from the offer text when
        // structured prices are absent (most coupon-type offers) — only
        // when the text gives us BOTH a percent-off AND a clear final/
        // starting price to back-calculate from; anything vaguer is
        // skipped rather than risk posting made-up numbers.
        let orig = offer.original_price;
        let deal = offer.discount_price;
        let isEstimated = false;

        const hasRealPrices = orig != null && deal != null
          && !isNaN(Number(orig)) && !isNaN(Number(deal))
          && Number(deal) < Number(orig);

        if (!hasRealPrices) {
          const extracted = extractPriceFromText_(`${title} ${description}`);
          if (!extracted) continue; // couldn't confidently derive prices, skip
          orig = extracted.original;
          deal = extracted.deal;
          isEstimated = true;
        }

        if (existingLinks.has(link.trim())) continue;

        const haystack = `${title} ${description}`.toLowerCase();
        const category = classifyCategory_(haystack);
        if (!category) continue;

        if (quotaUsed[category] >= quotaLimit) continue;

        const displayName = isEstimated ? `${title.trim()} (Est.)` : title.trim();
        rowsToAppend.push([displayName, link.trim(), Number(orig), Number(deal), category, 'Pending']);
        quotaUsed[category]++;
        existingLinks.add(link.trim());

      } catch (err) {
        logError_(`fetchCuelinksOffers page ${page} item`, err);
      }
    }

    page++;
  }

  if (rowsToAppend.length > 0) {
    const startRow = sheet.getLastRow() + 1;
    sheet.getRange(startRow, 1, rowsToAppend.length, 6).setValues(rowsToAppend);
  }

  Logger.log(`fetchCuelinksOffers: appended ${rowsToAppend.length} rows across ${page - 1} page(s). Quota used: ${JSON.stringify(quotaUsed)} / limit ${quotaLimit}`);
}

/**
 * Generic intake engine: pulls one source's XML deals feed, de-dupes
 * against existing Raw_URL history, keyword-categorizes each item, and
 * appends fresh "Pending" rows up to a per-category quota per run.
 *
 * Does NOT touch processRow_ or the Telegram posting logic — this only
 * writes rows; the existing onEdit/sweep triggers pick them up normally.
 */
function fetchDealsFromSource_(source) {
  const sheet = SpreadsheetApp.getActive().getSheetByName(SHEET_NAME);

  if (!source.feedUrl || source.feedUrl.indexOf('PASTE_') === 0) {
    logError_(`fetchDealsFromSource_ (${source.name})`, new Error('feedUrl not configured'));
    return;
  }

  // --- Fetch ---
  let xmlText;
  try {
    const response = UrlFetchApp.fetch(source.feedUrl, { muteHttpExceptions: true });
    const code = response.getResponseCode();
    if (code !== 200) {
      throw new Error(`Feed fetch failed with HTTP ${code}`);
    }
    xmlText = response.getContentText();
  } catch (err) {
    logError_(`fetchDealsFromSource_ (${source.name}) fetch`, err);
    return;
  }

  // --- Parse ---
  let items;
  try {
    const document = XmlService.parse(xmlText);
    const root = document.getRootElement();
    items = findAllDescendants_(root, source.tags.itemNode);
  } catch (err) {
    logError_(`fetchDealsFromSource_ (${source.name}) parse`, err);
    return;
  }

  if (!items || items.length === 0) {
    logError_(`fetchDealsFromSource_ (${source.name})`, new Error('Feed parsed but returned 0 items — check tags.itemNode matches this feed'));
    return;
  }

  // --- Existing Raw_URL history, for de-dup (re-read fresh per source
  // so a Cuelinks item just written in this same run blocks a duplicate
  // from EarnKaro later in the same run) ---
  const existingLinks = getExistingRawUrls_(sheet);

  // --- Quota tracking ---
  const quotaLimit = SYSTEM_SETTINGS.IS_SALE_MODE_ACTIVE
    ? INTAKE_QUOTA.saleMode
    : INTAKE_QUOTA.normal;
  const quotaUsed = { 'Gym': 0, 'Pets': 0, 'Vehicle': 0 };

  const rowsToAppend = [];

  for (const item of items) {
    const allFull = Object.keys(quotaUsed).every(cat => quotaUsed[cat] >= quotaLimit);
    if (allFull) break;

    try {
      const title = getChildText_(item, source.tags.titleTag);
      const link = getChildText_(item, source.tags.linkTag);
      const description = getChildText_(item, source.tags.descriptionTag);

      if (!title || !link) continue; // malformed item, skip silently

      if (existingLinks.has(link.trim())) continue;

      const haystack = `${title} ${description || ''}`.toLowerCase();
      const category = classifyCategory_(haystack);
      if (!category) continue;

      if (quotaUsed[category] >= quotaLimit) continue;

      rowsToAppend.push([title.trim(), link.trim(), '', '', category, 'Pending']);
      quotaUsed[category]++;
      existingLinks.add(link.trim());

    } catch (err) {
      logError_(`fetchDealsFromSource_ (${source.name}) item`, err);
    }
  }

  // --- Write ---
  if (rowsToAppend.length > 0) {
    const startRow = sheet.getLastRow() + 1;
    sheet.getRange(startRow, 1, rowsToAppend.length, 6).setValues(rowsToAppend);
  }

  Logger.log(`fetchDealsFromSource_ (${source.name}): appended ${rowsToAppend.length} rows. Quota used: ${JSON.stringify(quotaUsed)} / limit ${quotaLimit}`);
}

/**
 * Precompiled word-boundary regexes for CATEGORY_KEYWORDS, built once.
 * Plain substring matching (text.indexOf(kw)) was matching "cat" inside
 * "category" and "car" inside "card" — this uses \b boundaries so a
 * keyword only matches as a standalone word, not as a substring of an
 * unrelated word.
 */
function buildCategoryRegexes_() {
  const map = {};
  for (const category in CATEGORY_KEYWORDS) {
    map[category] = CATEGORY_KEYWORDS[category].map(
      kw => new RegExp('\\b' + kw.replace(/[.*+?^${}()|[\]\\]/g, '\\$&') + '\\b', 'i')
    );
  }
  return map;
}
const CATEGORY_KEYWORD_REGEXES = buildCategoryRegexes_();

/**
 * Matches text against CATEGORY_KEYWORDS using word-boundary regexes.
 * Returns the first matching category name, or null if no keyword hits.
 */
function classifyCategory_(text) {
  for (const category in CATEGORY_KEYWORD_REGEXES) {
    for (const re of CATEGORY_KEYWORD_REGEXES[category]) {
      if (re.test(text)) return category;
    }
  }
  return null;
}

/**
 * Fallback price extraction for offers with no structured original_price/
 * discount_price (most Cuelinks coupon-type offers). Only succeeds when
 * the text gives us BOTH a percent-off figure AND a clear final/starting
 * price ("only pay Rs. X", "starting at Rs. X", "prices from Rs. X",
 * "just Rs. X") — enough to back-calculate original = deal / (1 - pct/100).
 *
 * Deliberately conservative: text like "flat Rs. 200 off" with no base
 * price, or a lone percent with no price at all, returns null rather than
 * guessing. Returns { original, deal } (both rounded to 2dp) or null.
 */
function extractPriceFromText_(text) {
  const percentMatch = text.match(/(\d{1,3})\s*%\s*(?:off|discount|extra)?/i);
  if (!percentMatch) return null;

  const pct = Number(percentMatch[1]);
  if (isNaN(pct) || pct <= 0 || pct >= 100) return null;

  // Only trust a price figure when it's clearly framed as the resulting/
  // starting price, not a discount amount ("flat Rs. X off" is NOT this).
  const priceMatch = text.match(
    /(?:only pay|starting at|starts at|prices? (?:from|start(?:ing)? (?:at|from))|just)\s*(?:rs\.?|inr|₹)\s*([\d,]+(?:\.\d+)?)/i
  );
  if (!priceMatch) return null;

  const deal = Number(priceMatch[1].replace(/,/g, ''));
  if (isNaN(deal) || deal <= 0) return null;

  const original = deal / (1 - pct / 100);
  if (!isFinite(original) || original <= deal) return null;

  return {
    original: Math.round(original * 100) / 100,
    deal: Math.round(deal * 100) / 100
  };
}

/**
 * Reads every existing Raw_URL in the sheet into a Set, for O(1) de-dup
 * lookups instead of re-scanning the sheet per feed item.
 */
function getExistingRawUrls_(sheet) {
  const lastRow = sheet.getLastRow();
  const set = new Set();
  if (lastRow <= HEADER_ROW) return set;

  const links = sheet.getRange(HEADER_ROW + 1, COL.LINK, lastRow - HEADER_ROW, 1).getValues();
  links.forEach(row => {
    const val = String(row[0]).trim();
    if (val) set.add(val);
  });
  return set;
}

/**
 * XmlService has no built-in "getElementsByTagName"-style deep search,
 * so this walks the tree recursively collecting every element whose
 * local name matches tagName, at any depth.
 */
function findAllDescendants_(element, tagName) {
  let results = [];
  const children = element.getChildren();
  for (const child of children) {
    if (child.getName() === tagName) {
      results.push(child);
    }
    results = results.concat(findAllDescendants_(child, tagName));
  }
  return results;
}

/**
 * Safely reads a child element's text content, at any depth under the
 * item node (handles both direct children and namespaced/nested tags).
 */
function getChildText_(itemElement, tagName) {
  const direct = itemElement.getChild(tagName);
  if (direct) return direct.getText();

  const nested = findAllDescendants_(itemElement, tagName);
  return nested.length > 0 ? nested[0].getText() : '';
}


/**
 * Installable onEdit trigger — fires per edit, filters to real data rows.
 */
function onEditInstallable(e) {
  const __trackStart = Date.now();
  try {
    const sheet = e.range.getSheet();
    if (sheet.getName() !== SHEET_NAME) return;

    const editedRow = e.range.getRow();
    if (editedRow <= HEADER_ROW) return;

    processRow_(sheet, editedRow);
  } catch (err) {
    logError_('onEditInstallable', err);
  } finally {
    recordExecutionRuntime_('onEditInstallable', __trackStart);
  }
}

/**
 * Time-based safety net — sweeps for any row still marked Pending.
 * Wire to a time-driven trigger (createTrigger sets this to every 5 min).
 */
function sweepPendingRows() {
  const __trackStart = Date.now();
  try {
    const sheet = SpreadsheetApp.getActive().getSheetByName(SHEET_NAME);
    const lastRow = sheet.getLastRow();
    for (let row = HEADER_ROW + 1; row <= lastRow; row++) {
      try {
        processRow_(sheet, row);
      } catch (err) {
        logError_('sweepPendingRows row ' + row, err);
      }
    }
  } finally {
    recordExecutionRuntime_('sweepPendingRows', __trackStart);
  }
}

/**
 * Core per-row logic: validate -> classify -> format -> send -> update status.
 * Only acts on rows whose Status is exactly "Pending". Never throws out of
 * this function — every failure path writes "Error: <reason>" to column F
 * and returns, so one bad row never halts the sweep loop.
 */
function processRow_(sheet, row) {
  const values = sheet.getRange(row, 1, 1, 6).getValues()[0];
  const [name, link, origPrice, dealPrice, category, status] = values;

  // Only process rows explicitly marked Pending
  if (status !== 'Pending') return;

  // Ignore fully blank rows silently
  if (!name && !link && !origPrice && !dealPrice && !category) return;

  try {
    // --- Validation ---
    const missing = [];
    if (!name) missing.push('Product_Name');
    if (!link) missing.push('Raw_URL');
    if (!origPrice) missing.push('Original_Price');
    if (!dealPrice) missing.push('Deal_Price');
    if (!category) missing.push('Category');
    if (missing.length > 0) {
      throw new Error('Missing field(s): ' + missing.join(', '));
    }

    const orig = Number(origPrice);
    const deal = Number(dealPrice);
    if (isNaN(orig) || orig <= 0) throw new Error('Invalid Original_Price');
    if (isNaN(deal) || deal <= 0) throw new Error('Invalid Deal_Price');
    if (deal >= orig) throw new Error('Deal_Price must be lower than Original_Price');

    const categoryConfig = CATEGORY_MAP[category];
    if (!categoryConfig) {
      throw new Error(`Undefined category "${category}" (expected Gym, Pets, or Vehicle)`);
    }

    const props = PropertiesService.getScriptProperties();
    const botToken = props.getProperty('BOT_TOKEN');
    const chatId = props.getProperty(categoryConfig.propKey);

    if (!botToken) throw new Error('BOT_TOKEN not set in Script Properties');
    if (!chatId) throw new Error(`${categoryConfig.propKey} not set in Script Properties`);

    // --- Formatting ---
    const savingsPct = Math.round(((orig - deal) / orig) * 100);
    const message = buildMessage_(categoryConfig, name, orig, deal, savingsPct, link);

    // --- Send ---
    const result = sendToTelegram_(botToken, chatId, message, link);
    if (!result.ok) throw new Error(result.error);

    // --- Success ---
    sheet.getRange(row, COL.STATUS).setValue('Sent');

  } catch (err) {
    sheet.getRange(row, COL.STATUS).setValue('Error: ' + err.message);
    logError_('processRow_ row ' + row, err);
    // No re-throw — loop continues to the next row
  }
}

/**
 * Builds the MarkdownV2 message. MarkdownV2 requires escaping a specific
 * set of special characters in any user-supplied text (product names,
 * URLs can go in link syntax unescaped inside the parentheses).
 *
 * The Buy Now link is embedded as a Markdown hyperlink (short visible
 * text, full URL underneath) rather than pasted raw. This keeps the
 * message compact while still giving Telegram a URL in the message text
 * to generate a link preview (product image) from — inline keyboard
 * buttons alone are NOT scanned for previews, only text/text-link
 * entities in the message body are.
 */
function buildMessage_(categoryConfig, name, orig, deal, savingsPct, link) {
  const safeName = escapeMarkdownV2_(name);
  const origStr = escapeMarkdownV2_(`₹${orig.toLocaleString('en-IN')}`);
  const dealStr = escapeMarkdownV2_(`₹${deal.toLocaleString('en-IN')}`);
  const pctStr = escapeMarkdownV2_(`${savingsPct}%`);
  // Only the display text needs escaping; the URL inside () is left as-is
  // per MarkdownV2 spec, aside from escaping ) and \ if present in the URL.
  const safeLinkUrl = String(link).replace(/([)\\])/g, '\\$1');

  return [
    `${categoryConfig.emoji} *${categoryConfig.label}* ${categoryConfig.emoji}`,
    ``,
    `*${safeName}*`,
    ``,
    `~${origStr}~  ➡️  *${dealStr}*`,
    `🔥 *${pctStr} OFF*`,
    ``,
    `🛒 [Buy Now](${safeLinkUrl})`
  ].join('\n');
}

/**
 * Escapes MarkdownV2 reserved characters per Telegram's spec:
 * _ * [ ] ( ) ~ ` > # + - = | { } . !
 */
function escapeMarkdownV2_(text) {
  return String(text).replace(/[_*\[\]()~`>#+\-=|{}.!]/g, '\\$&');
}

/**
 * Sends the message to the resolved chat ID, with an inline "Buy Now"
 * button as the CTA. Retries on 5xx/timeouts and respects 429 retry_after.
 */
function sendToTelegram_(botToken, chatId, text, link) {
  const url = `https://api.telegram.org/bot${botToken}/sendMessage`;

  const payload = {
    chat_id: chatId,
    text: text,
    parse_mode: 'MarkdownV2',
    disable_web_page_preview: false,
    reply_markup: JSON.stringify({
      inline_keyboard: [[{ text: '🛒 Buy Now', url: link }]]
    })
  };

  const options = {
    method: 'post',
    contentType: 'application/x-www-form-urlencoded',
    payload: payload,
    muteHttpExceptions: true
  };

  const maxAttempts = 3;
  let lastError = '';

  for (let attempt = 1; attempt <= maxAttempts; attempt++) {
    try {
      const response = UrlFetchApp.fetch(url, options);
      const code = response.getResponseCode();
      const body = JSON.parse(response.getContentText());

      if (code === 200 && body.ok) {
        return { ok: true };
      }

      lastError = `HTTP ${code}: ${body.description || 'unknown error'}`;

      if (code === 429 && body.parameters && body.parameters.retry_after) {
        Utilities.sleep(body.parameters.retry_after * 1000);
        continue;
      }

      if (code >= 500) {
        Utilities.sleep(attempt * 1000);
        continue;
      }

      break; // non-retryable (bad token, chat not found, malformed markdown, etc.)
    } catch (err) {
      lastError = err.message;
      Utilities.sleep(attempt * 1000);
    }
  }

  return { ok: false, error: lastError };
}

/**
 * One-time setup: installs the onEdit trigger + 5-minute sweep trigger.
 */
function createTrigger() {
  const ss = SpreadsheetApp.getActive();

  ScriptApp.getProjectTriggers().forEach(t => {
    if (t.getHandlerFunction() === 'onEditInstallable' ||
        t.getHandlerFunction() === 'sweepPendingRows') {
      ScriptApp.deleteTrigger(t);
    }
  });

  ScriptApp.newTrigger('onEditInstallable')
    .forSpreadsheet(ss)
    .onEdit()
    .create();

  ScriptApp.newTrigger('sweepPendingRows')
    .timeBased()
    .everyMinutes(5)
    .create();

  Logger.log('Triggers created.');
}

/**
 * Logs failures to a dedicated "Logs" tab for visibility without digging
 * through the Apps Script execution console.
 */
function logError_(context, err) {
  Logger.log(`[${context}] ${err.message}`);
  try {
    const ss = SpreadsheetApp.getActive();
    let logSheet = ss.getSheetByName('Logs');
    if (!logSheet) {
      logSheet = ss.insertSheet('Logs');
      logSheet.appendRow(['Timestamp', 'Context', 'Error']);
    }
    logSheet.appendRow([new Date(), context, err.message]);
  } catch (e) {
    // swallow logging errors so they never break the main flow
  }
}
